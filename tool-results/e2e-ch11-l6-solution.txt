import re
log = [
    'From stephen.marquard@uct.ac.za Sat Jan  5 09:14:16 2008',
    'X-DSPAM-Confidence: 0.8475',
    'Author: stephen.marquard@uct.ac.za',
    'X-DSPAM-Probability: 0.0000',
    'From cwen@iupui.edu Thu Jan  3 16:23:48 2008',
    'X-DSPAM-Confidence: 0.6178',
    'X-DSPAM-Probability: 0.0000',
    'X-DSPAM-Confidence: 0.6961',
    'Subject: Past deadline',
    'X-DSPAM-Confidence: 0.8386',
    'From zqian@umich.edu Fri Jan  4 18:10:48 2008',
]
total = 0.0
count = 0
for line in log:
    x = re.findall(r'^X-DSPAM-Confidence: ([0-9.]+)', line)
    if len(x) > 0:
        total = total + float(x[0])
        count = count + 1
print('confidence count: %d' % count)
print('average: %.4f' % (total / count))
