import ast

SAMPLE = '''def load(path):
    # TODO: validate the path before opening it
    try:
        return open(path).read()
    except:
        return ''
    # a long padded note aaaaaa bbbbbb cccccc dddddd eeeeee ffffff gggggg hhhhhh
    # another long padded note iiiiii jjjjjj kkkkkk llllll mmmmmm nnnnnn oooooo pppppp qqqqqq
'''

def lint(source):
    tree = ast.parse(source)
    bare = 0
    for node in ast.walk(tree):
        if isinstance(node, ast.ExceptHandler) and node.type is None:
            bare = bare + 1
    long_lines = []
    todos = 0
    for number, line in enumerate(source.split('\n'), start=1):
        if len(line) > 79:
            long_lines.append(number)
        todos = todos + line.count('TODO') + line.count('FIXME')
    return {'bare': bare, 'long': long_lines, 'todos': todos}


report = lint(SAMPLE)
print('bare except: %d' % report['bare'])
print('long lines: %d' % len(report['long']))
print('long at: %s' % report['long'])
print('todos: %d' % report['todos'])
